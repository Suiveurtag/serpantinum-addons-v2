cmake_minimum_required(VERSION 3.21)
project(SerpantinumWallpaper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.12 REQUIRED COMPONENTS Quick OpenGL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GPU REQUIRED IMPORTED_TARGET epoxy gbm egl)
qt_standard_project_setup()
qt_policy(SET QTP0001 NEW)
qt_add_qml_module(SerpantinumWallpaper URI Serpantinum.Wallpaper VERSION 1.0
    SOURCES texture.cpp texture.h protocol.h)
target_link_libraries(SerpantinumWallpaper PRIVATE Qt6::Quick Qt6::OpenGL PkgConfig::GPU)
add_library(we-glass-export SHARED export.cpp)
target_link_libraries(we-glass-export PRIVATE PkgConfig::GPU)
enable_testing()
add_executable(glass-protocol-test protocol_test.cpp)
target_compile_options(glass-protocol-test PRIVATE -UNDEBUG)
add_test(NAME glass-protocol COMMAND glass-protocol-test)
install(TARGETS we-glass-export LIBRARY DESTINATION lib)
install(TARGETS SerpantinumWallpaper SerpantinumWallpaperplugin LIBRARY DESTINATION lib/qt6/qml/Serpantinum/Wallpaper)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/qmldir ${CMAKE_CURRENT_BINARY_DIR}/SerpantinumWallpaper.qmltypes
    DESTINATION lib/qt6/qml/Serpantinum/Wallpaper)

add_executable(glass-frame-probe probe.cpp)
target_link_libraries(glass-frame-probe PRIVATE Qt6::Gui PkgConfig::GPU)

add_executable(glass-pool-test export_test.cpp)
target_compile_options(glass-pool-test PRIVATE -UNDEBUG)
target_link_libraries(glass-pool-test PRIVATE we-glass-export PkgConfig::GPU)
add_test(NAME glass-pool COMMAND glass-pool-test)
set_tests_properties(glass-pool PROPERTIES SKIP_RETURN_CODE 77)
